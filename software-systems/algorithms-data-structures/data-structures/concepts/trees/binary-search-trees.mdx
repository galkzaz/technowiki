A **Binary Search Tree (BST)** is a special type of binary tree that maintains an ordered structure, making searching, insertion, and deletion efficient.

A **Binary Search Tree** is a special type of Tree that can be efficiently searched. Nodes in Binary Search Trees can have at most two children. And nodes are positioned according to their value/key. Children nodes to the left of the parent must be smaller than the parent,
children nodes to the right must be greater.

## Definition
A BST is a binary tree where for every node:
- All values in the left subtree are less than the node’s value 
- All values in the right subtree are greater than the node’s value 

This rule must hold recursively for every node.

## Example

![A sample binary search tree.](/docs/assets/software-systems/data-structures-and-algorithms/data-structures/trees//sample-binary-tree.png)

```
        50
       /  \
     30    70
    / \    / \
  20  40  60  80
```
- Left of 50 → all values < 50 
- Right of 50 → all values > 50 
- Same rule applies to every subtree 

A BST is basically:  **Binary Tree + Ordering Constraint**, This constraint is what makes it powerful.
## Properties
Core Property (Sorted Structure)

If you traverse a BST using Inorder Traversal (Left → Root → Right), you always get: Sorted order

Example:
```
        50
       /  \
     30    70
    / \    / \
  20  40  60  80
```
Inorder → 20, 30, 40, 50, 60, 70, 80

If the tree respects this property, it’s easy to search for a node with a given key/value within the tree(see search operation bellow).
## Basic Operations
### Search
Idea: Compare and move left or right.

Algorithm:
```
if root == NULL → not found
if key == root → found
if key < root → search left
if key > root → search right
```
Pseudocode
```
function find_node(binary_tree, value)
  node ← binary_tree.root_node
  while node:
    if node.value = value
      return node
    if value > node.value
      node ← node.right
    else
      node ← node.left
  return "NOT FOUND"
```
Time Complexity:
- Best / Average: O(log n) 
- Worst (skewed tree): O(n) 

### Insertion
Rule: Insert like search until you find a NULL position.

To insert an item, we search the value we want to insert in the tree.
We take the last node explored in that search, and make its right
or left pointer point to the new node:
Pseudocode
```
function insert_node(binary_tree, new_node)
  node ← binary_tree.root_node
  while node:
    last_node ← node
    if new_node.value > node.value
      node ← node.right
    else
      node ← node.left
  if new_node.value > last_node.value
    last_node.right ← new_node
  else
    last_node.left ← new_node
```
Example:
```
Insert 65:
50 → 70 → 60 → insert as right child
```
Result:
```
        50
       /  \
     30    70
           /
         60
           \
           65

```
### Deletion
Three cases:
1. **Case 1**: Leaf node
    - Just remove it.
2. **Case 2**: One child
    - Replace node with its child.
3. **Case 3**: Two children
    - Replace node with:
        - Inorder successor (smallest in right subtree), OR 
        - Inorder predecessor (largest in left subtree) 

Pseudocode
```
function delete_node(binary_tree, value)
  node ← binary_tree.root_node
  parent ← null

  // Find node to delete
  while node ≠ null and node.value ≠ value:
    parent ← node
    if value < node.value
      node ← node.left
    else
      node ← node.right

  if node = null
    return "NOT FOUND"

  // Case 1: Node has no children
  if node.left = null and node.right = null
    if node = parent.left
      parent.left ← null
    else
      parent.right ← null
    delete node
    return

  // Case 2: Node has one child
  if node.left = null or node.right = null
    child ← node.left ? node.left : node.right
    if node = parent.left
      parent.left ← child
    else
      parent.right ← child
    delete node
    return

  // Case 3: Node has two children
  successor ← find_min(node.right)
  value_to_replace ← successor.value
  delete_node(binary_tree, successor.value)
  node.value ← value_to_replace
```
### Find Minimum / Maximum
Pseudocode
```
function find_min(node)
  while node.left ≠ null
    node ← node.left
  return node

function find_max(node)
  while node.right ≠ null
    node ← node.right
  return node
```

function inorder(node)

Pseudocode

Pseudocode

function postorder(node)

```

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// 1. Node Creation: Structure for a BST Node
typedef struct Node {
    int data;
    struct Node *left, *right;
} Node;

Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->left = newNode->right = NULL;
    return newNode;
}

// 2. Insertion: Recursive insertion
Node* insert(Node* root, int data) {
    if (root == NULL) return createNode(data);
    if (data < root->data)
        root->left = insert(root->left, data);
    else if (data > root->data)
        root->right = insert(root->right, data);
    return root;
}

// 3. Traversals
void preorder(Node* root) {
    if (root) {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

void inorder(Node* root) {
    if (root) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

void postorder(Node* root) {
    if (root) {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

// Level Order (Helper for height-based level printing)
void printCurrentLevel(Node* root, int level) {
    if (root == NULL) return;
    if (level == 1) printf("%d ", root->data);
    else if (level > 1) {
        printCurrentLevel(root->left, level - 1);
        printCurrentLevel(root->right, level - 1);
    }
}

// 5. Get Height
int getHeight(Node* node) {
    if (node == NULL) return 0;
    int lHeight = getHeight(node->left);
    int rHeight = getHeight(node->right);
    return (lHeight > rHeight) ? (lHeight + 1) : (rHeight + 1);
}

void levelOrder(Node* root) {
    int h = getHeight(root);
    for (int i = 1; i <= h; i++) printCurrentLevel(root, i);
}

// 4. Search
Node* search(Node* root, int key) {
    if (root == NULL || root->data == key) return root;
    if (key < root->data) return search(root->left, key);
    return search(root->right, key);
}

// 6. Delete Entire Tree (Memory Free)
void deleteTree(Node* root) {
    if (root == NULL) return;
    deleteTree(root->left);
    deleteTree(root->right);
    free(root);
}

// 7. Count Nodes
int countNodes(Node* root) {
    if (root == NULL) return 0;
    return 1 + countNodes(root->left) + countNodes(root->right);
}

// 8. Count Leaf Nodes
int countLeafNodes(Node* root) {
    if (root == NULL) return 0;
    if (root->left == NULL && root->right == NULL) return 1;
    return countLeafNodes(root->left) + countLeafNodes(root->right);
}

// 9. Find Max / Min
int findMin(Node* root) {
    Node* current = root;
    while (current && current->left != NULL) current = current->left;
    return current->data;
}

int findMax(Node* root) {
    Node* current = root;
    while (current && current->right != NULL) current = current->right;
    return current->data;
}

// 10. Copy Tree
Node* copyTree(Node* root) {
    if (root == NULL) return NULL;
    Node* newNode = createNode(root->data);
    newNode->left = copyTree(root->left);
    newNode->right = copyTree(root->right);
    return newNode;
}

// 11. Tree Equality
bool isSameTree(Node* t1, Node* t2) {
    if (t1 == NULL && t2 == NULL) return true;
    if (t1 != NULL && t2 != NULL) {
        return (t1->data == t2->data && 
                isSameTree(t1->left, t2->left) && 
                isSameTree(t1->right, t2->right));
    }
    return false;
}

// 12. Mirror Tree
void mirror(Node* root) {
    if (root == NULL) return;
    mirror(root->left);
    mirror(root->right);
    Node* temp = root->left;
    root->left = root->right;
    root->right = temp;
}

// 13. Check Balanced
bool isBalanced(Node* root) {
    if (root == NULL) return true;
    int lh = getHeight(root->left);
    int rh = getHeight(root->right);
    return abs(lh - rh) <= 1 && isBalanced(root->left) && isBalanced(root->right);
}

// 14. Diameter
int max(int a, int b) { return (a > b) ? a : b; }
int diameter(Node* root) {
    if (root == NULL) return 0;
    int lh = getHeight(root->left);
    int rh = getHeight(root->right);
    int ldiameter = diameter(root->left);
    int rdiameter = diameter(root->right);
    return max(lh + rh + 1, max(ldiameter, rdiameter));
}

int main() {
    Node* root = NULL;
    root = insert(root, 50);
    insert(root, 30); insert(root, 70);
    insert(root, 20); insert(root, 40);

    printf("Inorder: "); inorder(root); printf("\n");
    printf("Height: %d\n", getHeight(root));
    printf("Nodes: %d, Leaves: %d\n", countNodes(root), countLeafNodes(root));
    printf("Diameter: %d\n", diameter(root));

    deleteTree(root);
    return 0;
}
```

**Notes:**
- **Memory Management**: In C, you must manually walk the tree to free nodes using a post-order traversal to avoid memory leaks.
- **Level Order**: Implemented here using a recursive height-based approach for simplicity; an explicit Queue is more efficient for large trees.
- **Complexity**: Most operations are $O(h)$, where $h$ is height. In the worst case (skewed tree), $h = n$.

#include <iostream>
#include <algorithm>
#include <queue>

using namespace std;

// 1. Node Creation
struct Node {
    int data;
    Node *left, *right;
    Node(int val) : data(val), left(nullptr), right(nullptr) {}
};

class BST {
public:
    Node* root;
    BST() : root(nullptr) {}

    // 2. Insertion
    void insert(int data) { root = insertRec(root, data); }

    // 3. Traversals
    void inorder() { inorderRec(root); cout << endl; }
    void levelOrder() {
        if (!root) return;
        queue<Node*> q;
        q.push(root);
        while (!q.empty()) {
            Node* curr = q.front(); q.pop();
            cout << curr->data << " ";
            if (curr->left) q.push(curr->left);
            if (curr->right) q.push(curr->right);
        }
        cout << endl;
    }

    // 5. Height
    int getHeight(Node* node) {
        if (!node) return 0;
        return 1 + max(getHeight(node->left), getHeight(node->right));
    }

    // 7. Count Nodes
    int countNodes(Node* node) {
        if (!node) return 0;
        return 1 + countNodes(node->left) + countNodes(node->right);
    }

    // 12. Mirror
    void mirror(Node* node) {
        if (!node) return;
        swap(node->left, node->right);
        mirror(node->left);
        mirror(node->right);
    }

    // 14. Diameter
    int getDiameter(Node* node) {
        if (!node) return 0;
        int h = getHeight(node->left) + getHeight(node->right) + 1;
        return max({h, getDiameter(node->left), getDiameter(node->right)});
    }

    // 6. Memory Cleanup
    void clear(Node* node) {
        if (!node) return;
        clear(node->left);
        clear(node->right);
        delete node;
    }

    ~BST() { clear(root); }

private:
    Node* insertRec(Node* node, int data) {
        if (!node) return new Node(data);
        if (data < node->data) node->left = insertRec(node->left, data);
        else if (data > node->data) node->right = insertRec(node->right, data);
        return node;
    }

    void inorderRec(Node* node) {
        if (!node) return;
        inorderRec(node->left);
        cout << node->data << " ";
        inorderRec(node->right);
    }
};

int main() {
    BST tree;
    tree.insert(50); tree.insert(30); tree.insert(70);
    cout << "Inorder Traversal: "; tree.inorder();
    cout << "Level Order: "; tree.levelOrder();
    cout << "Tree Height: " << tree.getHeight(tree.root) << endl;
    cout << "Tree Diameter: " << tree.getDiameter(tree.root) << endl;
    return 0;
}
```

**Notes:**
- **Standard Library**: C++ allows the use of `std::queue` for an efficient $O(n)$ level-order traversal.
- **Encapsulation**: Using a class to wrap the root node is the standard OOP approach.
- **Cleanup**: The destructor ensures the entire tree is deleted from the heap when the `BST` object goes out of scope.

import java.util.*;

class Node {
    int data;
    Node left, right;
    Node(int data) {
        this.data = data;
        left = right = null;
    }
}

public class BinarySearchTree {
    Node root;

    // 2. Insertion
    void insert(int data) { root = insertRec(root, data); }
    private Node insertRec(Node root, int data) {
        if (root == null) return new Node(data);
        if (data < root.data) root.left = insertRec(root.left, data);
        else if (data > root.data) root.right = insertRec(root.right, data);
        return root;
    }

    // 3. Traversals
    void levelOrder() {
        Queue<Node> queue = new LinkedList<>();
        queue.add(root);
        while (!queue.isEmpty()) {
            Node temp = queue.poll();
            System.out.print(temp.data + " ");
            if (temp.left != null) queue.add(temp.left);
            if (temp.right != null) queue.add(temp.right);
        }
    }

    // 5. Get Height
    int getHeight(Node node) {
        if (node == null) return 0;
        return 1 + Math.max(getHeight(node.left), getHeight(node.right));
    }

    // 11. Tree Equality
    static boolean isSameTree(Node p, Node q) {
        if (p == null && q == null) return true;
        if (p == null || q == null) return false;
        return (p.data == q.data) && isSameTree(p.left, q.left) && isSameTree(p.right, q.right);
    }

    // 13. Check Balanced
    boolean isBalanced(Node node) {
        if (node == null) return true;
        int lh = getHeight(node.left);
        int rh = getHeight(node.right);
        return Math.abs(lh - rh) <= 1 && isBalanced(node.left) && isBalanced(node.right);
    }

    public static void main(String[] args) {
        BinarySearchTree bst = new BinarySearchTree();
        bst.insert(50); bst.insert(30); bst.insert(70);
        System.out.print("Level Order: "); bst.levelOrder();
        System.out.println("\nHeight: " + bst.getHeight(bst.root));
        System.out.println("Is Balanced: " + bst.isBalanced(bst.root));
    }
}
```

- **Garbage Collection**: Unlike C/C++, Java handles memory deletion automatically. Setting `root = null` eventually triggers GC for all reachable nodes.
- **Queue Interface**: `LinkedList` is commonly used as a `Queue` implementation for Breadth-First Search (Level Order).

```javascript
class Node {
    constructor(data) {
        this.data = data;
        this.left = null;
        this.right = null;
    }
}

class BST {
    constructor() {
        this.root = null;
    }

    // 2. Insertion
    insert(data) {
        const newNode = new Node(data);
        if (!this.root) this.root = newNode;
        else this.insertNode(this.root, newNode);
    }

    insertNode(node, newNode) {
        if (newNode.data < node.data) {
            if (!node.left) node.left = newNode;
            else this.insertNode(node.left, newNode);
        } else {
            if (!node.right) node.right = newNode;
            else this.insertNode(node.right, newNode);
        }
    }

    // 3. Level Order Traversal
    levelOrder() {
        let res = [];
        let queue = [this.root];
        while (queue.length) {
            let node = queue.shift();
            if (node) {
                res.push(node.data);
                queue.push(node.left, node.right);
            }
        }
        return res;
    }

    // 5. Get Height
    getHeight(node = this.root) {
        if (!node) return 0;
        return 1 + Math.max(this.getHeight(node.left), this.getHeight(node.right));
    }

    // 12. Mirror Tree
    mirror(node = this.root) {
        if (!node) return;
        [node.left, node.right] = [node.right, node.left];
        this.mirror(node.left);
        this.mirror(node.right);
    }
}

// Testing
const tree = new BST();
[50, 30, 70, 20, 40].forEach(val => tree.insert(val));
console.log("Level Order:", tree.levelOrder());
console.log("Height:", tree.getHeight());
tree.mirror();
console.log("Mirrored Level Order:", tree.levelOrder());
```

**Notes:**
- **Dynamic Objects**: JavaScript makes tree manipulation very concise, especially using array destructuring for swapping nodes (Mirror).
- **Queue**: `Array.shift()` is used for the queue, though for very large trees, a proper Doubly Linked List queue would perform better ($O(1)$ vs $O(n)$ shift).

```rust
use std::cmp;

// 1. Node Creation
#[derive(Debug, Clone)]
struct Node {
    data: i32,
    left: Option<Box<Node>>,
    right: Option<Box<Node>>,
}

impl Node {
    fn new(data: i32) -> Self {
        Node { data, left: None, right: None }
    }
}

// 2. Insertion
fn insert(root: Option<Box<Node>>, data: i32) -> Option<Box<Node>> {
    match root {
        None => Some(Box::new(Node::new(data))),
        Some(mut node) => {
            if data < node.data {
                node.left = insert(node.left, data);
            } else if data > node.data {
                node.right = insert(node.right, data);
            }
            Some(node)
        }
    }
}

// 5. Get Height
fn get_height(root: &Option<Box<Node>>) -> i32 {
    match root {
        None => 0,
        Some(node) => 1 + cmp::max(get_height(&node.left), get_height(&node.right)),
    }
}

// 7. Count Nodes
fn count_nodes(root: &Option<Box<Node>>) -> i32 {
    match root {
        None => 0,
        Some(node) => 1 + count_nodes(&node.left) + count_nodes(&node.right),
    }
}

// 11. Tree Equality
fn is_same_tree(t1: &Option<Box<Node>>, t2: &Option<Box<Node>>) -> bool {
    match (t1, t2) {
        (None, None) => true,
        (Some(n1), Some(n2)) => {
            n1.data == n2.data && 
            is_same_tree(&n1.left, &n2.left) && 
            is_same_tree(&n1.right, &n2.right)
        },
        _ => false,
    }
}

fn main() {
    let mut root: Option<Box<Node>> = None;
    root = insert(root, 50);
    root = insert(root, 30);
    root = insert(root, 70);

    println!("Tree Height: {}", get_height(&root));
    println!("Node Count: {}", count_nodes(&root));
}
```

**Notes:**
- **Ownership**: Rust uses `Option<Box<Node>>` to handle nullable pointers safely. `Box` provides heap allocation.
- **Pattern Matching**: `match` is the idiomatic way to handle the recursive nature of trees in Rust.
- **Safety**: Rust's ownership model automatically frees the tree when the root variable goes out of scope, preventing memory leaks without a manual GC.

## Binary Search Tree (BST) vs a Heap
Binary Search Tree (BST) and Heap looks similar (both are binary trees), but behave very differently.

**Core Idea**
- BST → designed for searching 
- Heap → designed for priority access (min/max) 

Think:
- BST = “ordered dictionary” 
- Heap = “priority queue”

**Structural Difference**
| Feature | BST                                  | Heap                       |
| ------- | ------------------------------------ | -------------------------- |
| Shape   | Any shape (not necessarily complete) | **Complete binary tree**   |
| Balance | Can become skewed                    | Always balanced (by shape) |

**Ordering Rule**

**BST Rule**:
```
Left < Root < Right
```
Full ordering across the tree

**Heap Rule**:

Max Heap:
```
Parent ≥ Children
```
Min Heap:
```
Parent ≤ Children
```
Only local ordering, NOT global

**Example**
BST
```
        50
       /  \
     30    70
    / \    / \
  20  40  60  80
```
Fully sorted structure

Max Heap
```
        80
       /  \
     70    60
    / \    /
  20  50  40
```
Only parent is greater—not globally sorted

A Heap is NOT sorted, Example:
```
Heap array: [10, 20, 15, 30, 40]
```
- 10 is smallest (min heap)
- BUT 20 and 15 are not sorted relative to each other

**Traversal Behavior**
| Traversal   | BST         | Heap                     |
| ----------- | ----------- | ------------------------ |
| Inorder     | Sorted    | Not sorted             |
| Level Order | Not special | Natural representation |

**Time Complexity**
| Operation        | BST (avg) | Heap     |
| ---------------- | --------- | -------- |
| Search arbitrary | O(log n)  | **O(n)**   |
| Insert           | O(log n)  | O(log n) |
| Delete           | O(log n)  | O(log n) |
| Find Min/Max     | O(log n)  | **O(1)**   |

- Heap wins for min/max
- BST wins for search

**Implementation Difference**

BST
- Usually **pointer-based (linked nodes)** 
- Dynamic structure 

Heap
- Usually **array-based**

**Key Operations**

BST
- Search(key) 
- Insert(key) 
- Delete(key) 
- Traversals 

Heap
- Insert (heapify up) 
- Extract Min/Max (heapify down) 
- Peek (get root)

**Use Cases**

BST Used For:
- Fast lookup (like maps/sets) 
- Ordered data 
- Range queries 

Heap Used For:
- Priority Queue 
- Scheduling (OS, tasks) 
- Algorithms: 
    - Dijkstra 
    - Prim 
    - Heapsort

**Degeneration vs Stability**
| Issue                | BST   | Heap |
| -------------------- | ----- | ---- |
| Can degrade to O(n)? | ✅ Yes | ❌ No |
| Needs balancing?     | ✅ Yes | ❌ No |

**Summary**

BST
- Global ordering 
- Fast search 
- Can degrade 

Heap
- Partial ordering 
- Fast min/max 
- Always balanced (shape)

**Analogy**
- BST = Sorted phonebook → fast lookup by name 
- Heap = Urgent tasks list → always pick highest priority

**BST vs Heap in C implementation**

this comparison focus on insertion and extracting/searching, which are the core operations, so you can see the structural and behavioral differences clearly.

1️⃣ Node / Structure Definition
BST Node (Pointer-based)
```c
#include <stdio.h>
#include <stdlib.h>

typedef struct BSTNode {
    int key;
    struct BSTNode* left;
    struct BSTNode* right;
} BSTNode;

BSTNode* createBSTNode(int key) {
    BSTNode* node = (BSTNode*)malloc(sizeof(BSTNode));
    node->key = key;
    node->left = node->right = NULL;
    return node;
}
```
Heap (Array-based)
```c
#define MAX 100

typedef struct Heap {
    int size;
    int arr[MAX];
} Heap;

Heap* createHeap() {
    Heap* h = (Heap*)malloc(sizeof(Heap));
    h->size = 0;
    return h;
}
```
2️⃣ Insertion
BST Insertion (Recursive, maintains ordering)
```c
BSTNode* insertBST(BSTNode* root, int key) {
    if (root == NULL) return createBSTNode(key);

    if (key < root->key)
        root->left = insertBST(root->left, key);
    else if (key > root->key)
        root->right = insertBST(root->right, key);
    
    return root;
}
```
Heap Insertion (Max Heap, array-based)
```c
void swap(int* a, int* b) {
    int tmp = *a; *a = *b; *b = tmp;
}

void insertHeap(Heap* h, int key) {
    if (h->size >= MAX) return;
    int i = h->size++;
    h->arr[i] = key;

    // Heapify Up
    while (i != 0 && h->arr[i] > h->arr[(i-1)/2]) {
        swap(&h->arr[i], &h->arr[(i-1)/2]);
        i = (i-1)/2;
    }
}
```
3️⃣ Searching / Extracting

BST Search (O(log n) on average)
```x
BSTNode* searchBST(BSTNode* root, int key) {
    if (root == NULL || root->key == key) return root;
    if (key < root->key) return searchBST(root->left, key);
    return searchBST(root->right, key);
}
```
Heap Extract Max (O(log n))
```c
int extractMax(Heap* h) {
    if (h->size <= 0) return -1;
    int root = h->arr[0];
    h->arr[0] = h->arr[--h->size];

    // Heapify Down
    int i = 0;
    while (2*i + 1 < h->size) {
        int left = 2*i + 1;
        int right = 2*i + 2;
        int largest = i;

        if (left < h->size && h->arr[left] > h->arr[largest]) largest = left;
        if (right < h->size && h->arr[right] > h->arr[largest]) largest = right;
        if (largest == i) break;
        swap(&h->arr[i], &h->arr[largest]);
        i = largest;
    }
    return root;
}
```
4️⃣ Traversal / Print

BST Inorder Traversal (gives sorted order)
```c
void inorderBST(BSTNode* root) {
    if (!root) return;
    inorderBST(root->left);
    printf("%d ", root->key);
    inorderBST(root->right);
}
```
Heap Print (level-order representation)
```c
void printHeap(Heap* h) {
    for(int i=0; i<h->size; i++) printf("%d ", h->arr[i]);
    printf("\n");
}
```
5️⃣ Example Usage
```c
int main() {
    // BST
    BSTNode* root = NULL;
    root = insertBST(root, 50);
    insertBST(root, 30);
    insertBST(root, 70);
    insertBST(root, 20);
    insertBST(root, 40);
    printf("BST Inorder: ");
    inorderBST(root);  // 20 30 40 50 70
    printf("\n");

    // Heap
    Heap* h = createHeap();
    insertHeap(h, 50);
    insertHeap(h, 30);
    insertHeap(h, 70);
    insertHeap(h, 20);
    insertHeap(h, 40);
    printf("Heap Array: ");
    printHeap(h);  // 70 40 50 20 30 (Max Heap)
    printf("\n");
    
    printf("Extract Max from Heap: %d\n", extractMax(h)); // 70
    printHeap(h);  // Heap restructured
}
```
Key Observations from Side-by-Side:
| Feature   | BST                          | Heap                           |
| --------- | ---------------------------- | ------------------------------ |
| Structure | Pointer-based                | Array-based                    |
| Ordering  | Global (inorder = sorted)    | Local (parent ≥ children)      |
| Search    | O(log n)                     | O(n) for arbitrary element     |
| Insert    | O(log n) avg                 | O(log n)                       |
| Extract   | Delete node requires 3 cases | Always root (max/min) O(log n) |
| Traversal | Sorted output                | Level-order (not sorted)       |
