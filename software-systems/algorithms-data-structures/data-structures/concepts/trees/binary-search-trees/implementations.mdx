## C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

// 1. Node Creation: Structure for a BST Node
typedef struct Node {
    int data;
    struct Node *left, *right;
} Node;

Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->left = newNode->right = NULL;
    return newNode;
}

// 2. Insertion: Recursive insertion
Node* insert(Node* root, int data) {
    if (root == NULL) return createNode(data);
    if (data < root->data)
        root->left = insert(root->left, data);
    else if (data > root->data)
        root->right = insert(root->right, data);
    return root;
}

// 3. Traversals
void preorder(Node* root) {
    if (root) {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

void inorder(Node* root) {
    if (root) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

void postorder(Node* root) {
    if (root) {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

// Level Order (Helper for height-based level printing)
void printCurrentLevel(Node* root, int level) {
    if (root == NULL) return;
    if (level == 1) printf("%d ", root->data);
    else if (level > 1) {
        printCurrentLevel(root->left, level - 1);
        printCurrentLevel(root->right, level - 1);
    }
}

// 5. Get Height
int getHeight(Node* node) {
    if (node == NULL) return 0;
    int lHeight = getHeight(node->left);
    int rHeight = getHeight(node->right);
    return (lHeight > rHeight) ? (lHeight + 1) : (rHeight + 1);
}

void levelOrder(Node* root) {
    int h = getHeight(root);
    for (int i = 1; i <= h; i++) printCurrentLevel(root, i);
}

// 4. Search
Node* search(Node* root, int key) {
    if (root == NULL || root->data == key) return root;
    if (key < root->data) return search(root->left, key);
    return search(root->right, key);
}

// 6. Delete Entire Tree (Memory Free)
void deleteTree(Node* root) {
    if (root == NULL) return;
    deleteTree(root->left);
    deleteTree(root->right);
    free(root);
}

// 7. Count Nodes
int countNodes(Node* root) {
    if (root == NULL) return 0;
    return 1 + countNodes(root->left) + countNodes(root->right);
}

// 8. Count Leaf Nodes
int countLeafNodes(Node* root) {
    if (root == NULL) return 0;
    if (root->left == NULL && root->right == NULL) return 1;
    return countLeafNodes(root->left) + countLeafNodes(root->right);
}

// 9. Find Max / Min
int findMin(Node* root) {
    Node* current = root;
    while (current && current->left != NULL) current = current->left;
    return current->data;
}

int findMax(Node* root) {
    Node* current = root;
    while (current && current->right != NULL) current = current->right;
    return current->data;
}

// 10. Copy Tree
Node* copyTree(Node* root) {
    if (root == NULL) return NULL;
    Node* newNode = createNode(root->data);
    newNode->left = copyTree(root->left);
    newNode->right = copyTree(root->right);
    return newNode;
}

// 11. Tree Equality
bool isSameTree(Node* t1, Node* t2) {
    if (t1 == NULL && t2 == NULL) return true;
    if (t1 != NULL && t2 != NULL) {
        return (t1->data == t2->data && 
                isSameTree(t1->left, t2->left) && 
                isSameTree(t1->right, t2->right));
    }
    return false;
}

// 12. Mirror Tree
void mirror(Node* root) {
    if (root == NULL) return;
    mirror(root->left);
    mirror(root->right);
    Node* temp = root->left;
    root->left = root->right;
    root->right = temp;
}

// 13. Check Balanced
bool isBalanced(Node* root) {
    if (root == NULL) return true;
    int lh = getHeight(root->left);
    int rh = getHeight(root->right);
    return abs(lh - rh) <= 1 && isBalanced(root->left) && isBalanced(root->right);
}

// 14. Diameter
int max(int a, int b) { return (a > b) ? a : b; }
int diameter(Node* root) {
    if (root == NULL) return 0;
    int lh = getHeight(root->left);
    int rh = getHeight(root->right);
    int ldiameter = diameter(root->left);
    int rdiameter = diameter(root->right);
    return max(lh + rh + 1, max(ldiameter, rdiameter));
}

    Node* root = NULL;
    root = insert(root, 50);
    insert(root, 30); insert(root, 70);
    insert(root, 20); insert(root, 40);

    printf("Inorder: "); inorder(root); printf("\n");
    printf("Height: %d\n", getHeight(root));
    printf("Nodes: %d, Leaves: %d\n", countNodes(root), countLeafNodes(root));
    printf("Diameter: %d\n", diameter(root));

    return 0;
}
```

**Notes:**
- **Memory Management**: In C, you must manually walk the tree to free nodes using a post-order traversal to avoid memory leaks.
- **Level Order**: Implemented here using a recursive height-based approach for simplicity; an explicit Queue is more efficient for large trees.
- **Complexity**: Most operations are $O(h)$, where $h$ is height. In the worst case (skewed tree), $h = n$.

```cpp
    int getDiameter(Node* node) {
        if (!node) return 0;
        int h = getHeight(node->left) + getHeight(node->right) + 1;
        return max({h, getDiameter(node->left), getDiameter(node->right)});
    }

    void clear(Node* node) {
        if (!node) return;
        clear(node->left);
        clear(node->right);

        if (!node) return new Node(data);
        if (data < node->data) node->left = insertRec(node->left, data);
        else if (data > node->data) node->right = insertRec(node->right, data);
        return node;
```

- **Encapsulation**: Using a class to wrap the root node is the standard OOP approach.

## JavaScript
```javascript
class Node {
    constructor(data) {
        this.data = data;
        this.left = null;
        this.right = null;
    }
}

class BST {
    constructor() {
        this.root = null;
    }

    // 2. Insertion
    insert(data) {
        const newNode = new Node(data);
        if (!this.root) this.root = newNode;
        else this.insertNode(this.root, newNode);
    }

    insertNode(node, newNode) {
        if (newNode.data < node.data) {
            if (!node.left) node.left = newNode;
            else this.insertNode(node.left, newNode);
        } else {
            if (!node.right) node.right = newNode;
            else this.insertNode(node.right, newNode);
        }
    }

    // 3. Level Order Traversal
    levelOrder() {
        let res = [];
        let queue = [this.root];
        while (queue.length) {
            let node = queue.shift();
            if (node) {
                res.push(node.data);
                queue.push(node.left, node.right);
            }
        }
        return res;
    }

    // 5. Get Height
    getHeight(node = this.root) {
        if (!node) return 0;
        return 1 + Math.max(this.getHeight(node.left), this.getHeight(node.right));
    }

    // 12. Mirror Tree
    mirror(node = this.root) {
        if (!node) return;
        [node.left, node.right] = [node.right, node.left];
        this.mirror(node.left);
        this.mirror(node.right);
    }
}

// Testing
const tree = new BST();
[50, 30, 70, 20, 40].forEach(val => tree.insert(val));
console.log("Level Order:", tree.levelOrder());
console.log("Height:", tree.getHeight());
tree.mirror();
console.log("Mirrored Level Order:", tree.levelOrder());
```

**Notes:**
- **Dynamic Objects**: JavaScript makes tree manipulation very concise, especially using array destructuring for swapping nodes (Mirror).
- **Queue**: `Array.shift()` is used for the queue, though for very large trees, a proper Doubly Linked List queue would perform better ($O(1)$ vs $O(n)$ shift).

```rust
#[derive(Debug, Clone)]
struct Node {
    data: i32,
    left: Option<Box<Node>>,
    right: Option<Box<Node>>,

// 2. Insertion
fn insert(root: Option<Box<Node>>, data: i32) -> Option<Box<Node>> {
    match root {
        None => Some(Box::new(Node::new(data))),
        Some(mut node) => {
            if data < node.data {
                node.left = insert(node.left, data);
            } else if data > node.data {
                node.right = insert(node.right, data);
            }
            Some(node)
        }
    }
}

// 5. Get Height
fn get_height(root: &Option<Box<Node>>) -> i32 {
    match root {
        None => 0,
        Some(node) => 1 + cmp::max(get_height(&node.left), get_height(&node.right)),
    }
}

// 7. Count Nodes
fn count_nodes(root: &Option<Box<Node>>) -> i32 {
    match root {
        None => 0,
        Some(node) => 1 + count_nodes(&node.left) + count_nodes(&node.right),
    }
}

// 11. Tree Equality
fn is_same_tree(t1: &Option<Box<Node>>, t2: &Option<Box<Node>>) -> bool {
    match (t1, t2) {
        (None, None) => true,
        (Some(n1), Some(n2)) => {
            n1.data == n2.data && 
            is_same_tree(&n1.left, &n2.left) && 
            is_same_tree(&n1.right, &n2.right)
        },
        _ => false,
    }
}

}
```

**Notes:**
- **Ownership**: Rust uses `Option<Box<Node>>` to handle nullable pointers safely. `Box` provides heap allocation.
- **Pattern Matching**: `match` is the idiomatic way to handle the recursive nature of trees in Rust.
- **Safety**: Rust's ownership model automatically frees the tree when the root variable goes out of scope, preventing memory leaks without a manual GC.

## Python
Below is a complete Binary Search Tree (BST) implementation in Python covering the main BST operations.

The implementation uses a node-based BST and follows this rule:

- Left subtree: values smaller than the node.
- Right subtree: values greater than the node.
- Duplicate values are ignored.

```Python

class Node:
    """A single node in a Binary Search Tree."""

    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None


class BinarySearchTree:
    """Binary Search Tree with common operations."""

    def __init__(self):
        # An empty BST has no root.
        self.root = None

    # --------------------------------------------------
    # 1. INSERT
    # --------------------------------------------------
    def insert(self, value):
        """Insert a value. Ignore duplicates."""

        # The first inserted value becomes the root.
        if self.root is None:
            self.root = Node(value)
            return

        current = self.root

        while True:
            if value == current.value:
                # Duplicate values are not inserted.
                return

            elif value < current.value:
                # Smaller values belong in the left subtree.
                if current.left is None:
                    current.left = Node(value)
                    return
                current = current.left

            else:
                # Larger values belong in the right subtree.
                if current.right is None:
                    current.right = Node(value)
                    return
                current = current.right

    # --------------------------------------------------
    # 2. SEARCH
    # --------------------------------------------------
    def search(self, value):
        """Return the node containing value, or None."""

        current = self.root

        while current is not None:
            if value == current.value:
                return current

            elif value < current.value:
                current = current.left

            else:
                current = current.right

        # Reached an empty position: value does not exist.
        return None

    # --------------------------------------------------
    # 3. FIND MINIMUM
    # --------------------------------------------------
    def find_min_node(self, node):
        """Return the node with the smallest value in a subtree."""

        if node is None:
            return None

        # The smallest value is the leftmost node.
        while node.left is not None:
            node = node.left

        return node

    def find_min(self):
        """Return the smallest value, or None if empty."""

        node = self.find_min_node(self.root)
        return node.value if node is not None else None

    # --------------------------------------------------
    # 4. FIND MAXIMUM
    # --------------------------------------------------
    def find_max_node(self, node):
        """Return the node with the largest value in a subtree."""

        if node is None:
            return None

        # The largest value is the rightmost node.
        while node.right is not None:
            node = node.right

        return node

    def find_max(self):
        """Return the largest value, or None if empty."""

        node = self.find_max_node(self.root)
        return node.value if node is not None else None

    # --------------------------------------------------
    # 5. DELETE
    # --------------------------------------------------
    def delete(self, value):
        """Delete a value. Return True if found, otherwise False."""

        self.root, deleted = self._delete(self.root, value)
        return deleted

    def _delete(self, node, value):
        """Recursively delete a value from a subtree."""

        if node is None:
            # The value was not found.
            return None, False

        if value < node.value:
            # Search in the left subtree.
            node.left, deleted = self._delete(node.left, value)

        elif value > node.value:
            # Search in the right subtree.
            node.right, deleted = self._delete(node.right, value)

        else:
            # Found the node to delete.

            # Case 1: No children (leaf node).
            if node.left is None and node.right is None:
                return None, True

            # Case 2a: Only a right child.
            if node.left is None:
                return node.right, True

            # Case 2b: Only a left child.
            if node.right is None:
                return node.left, True

            # Case 3: Two children.
            # Find the inorder successor: the smallest value
            # in the right subtree.
            successor = self.find_min_node(node.right)

            # Copy the successor's value into this node.
            node.value = successor.value

            # Remove the original successor node.
            node.right, _ = self._delete(
                node.right, successor.value
            )

            return node, True

        return node, deleted

    # --------------------------------------------------
    # 6. INORDER TRAVERSAL: LEFT, ROOT, RIGHT
    # --------------------------------------------------
    def inorder(self):
        """Return values in ascending order."""

        result = []

        def traverse(node):
            if node is None:
                return

            traverse(node.left)
            result.append(node.value)
            traverse(node.right)

        traverse(self.root)
        return result

    # --------------------------------------------------
    # 7. PREORDER TRAVERSAL: ROOT, LEFT, RIGHT
    # --------------------------------------------------
    def preorder(self):
        """Return root before its subtrees."""

        result = []

        def traverse(node):
            if node is None:
                return

            result.append(node.value)
            traverse(node.left)
            traverse(node.right)

        traverse(self.root)
        return result

    # --------------------------------------------------
    # 8. POSTORDER TRAVERSAL: LEFT, RIGHT, ROOT
    # --------------------------------------------------
    def postorder(self):
        """Return root after its subtrees."""

        result = []

        def traverse(node):
            if node is None:
                return

            traverse(node.left)
            traverse(node.right)
            result.append(node.value)

        traverse(self.root)
        return result

    # --------------------------------------------------
    # 9. LEVEL-ORDER TRAVERSAL (BREADTH-FIRST)
    # --------------------------------------------------
    def level_order(self):
        """Return values level by level, using a queue."""

        if self.root is None:
            return []

        result = []
        queue = [self.root]
        index = 0

        # Process each node before its children.
        while index < len(queue):
            node = queue[index]
            index += 1
            result.append(node.value)

            if node.left is not None:
                queue.append(node.left)

            if node.right is not None:
                queue.append(node.right)

        return result

    # --------------------------------------------------
    # 10. HEIGHT
    # --------------------------------------------------
    def height(self):
        """
        Return tree height measured in edges.
        Empty tree = -1; leaf node = 0.
        """

        def calculate(node):
            if node is None:
                return -1

            left_height = calculate(node.left)
            right_height = calculate(node.right)

            return 1 + max(left_height, right_height)

        return calculate(self.root)

    # --------------------------------------------------
    # 11. COUNT NODES
    # --------------------------------------------------
    def count_nodes(self):
        """Return the total number of nodes."""

        def count(node):
            if node is None:
                return 0

            return 1 + count(node.left) + count(node.right)

        return count(self.root)

    # --------------------------------------------------
    # 12. CHECK IF EMPTY
    # --------------------------------------------------
    def is_empty(self):
        """Return True if the tree has no nodes."""

        return self.root is None

    # --------------------------------------------------
    # 13. CLEAR TREE
    # --------------------------------------------------
    def clear(self):
        """Remove all nodes by discarding the root reference."""

        self.root = None


# ------------------------------------------------------
# TEST CASES
# ------------------------------------------------------
if __name__ == "__main__":
    bst = BinarySearchTree()

    # Insert values.
    for value in [50, 30, 70, 20, 40, 60, 80]:
        bst.insert(value)

    # Expected tree:
    #
    #          50
    #         /  \
    #       30    70
    #      / \    / \
    #     20 40  60 80

    print("Inorder:   ", bst.inorder())
    print("Preorder:  ", bst.preorder())
    print("Postorder: ", bst.postorder())
    print("Level-order:", bst.level_order())

    # Search.
    node = bst.search(40)
    print("Search 40:", node.value if node else "NOT FOUND")

    node = bst.search(99)
    print("Search 99:", node.value if node else "NOT FOUND")

    # Minimum and maximum.
    print("Minimum:", bst.find_min())
    print("Maximum:", bst.find_max())

    # Tree information.
    print("Height:", bst.height())
    print("Node count:", bst.count_nodes())
    print("Is empty:", bst.is_empty())

    # Delete a leaf node.
    print("Delete 20:", bst.delete(20))
    print("After deleting 20:", bst.inorder())

    # Delete a node with one child.
    print("Delete 30:", bst.delete(30))
    print("After deleting 30:", bst.inorder())

    # Delete a node with two children.
    print("Delete 50:", bst.delete(50))
    print("After deleting 50:", bst.inorder())

    # Delete a value that does not exist.
    print("Delete 99:", bst.delete(99))

    # Clear the tree.
    bst.clear()
    print("After clear:", bst.inorder())
    print("Is empty:", bst.is_empty())
```
